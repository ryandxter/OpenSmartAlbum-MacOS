## Task 1 — Implement Compound SVG Mask Parser & ViewBox Normalizer

**File:** [`src/domain/shapes.ts`](file:///Users/chiio/VSCode/albumaker/src/domain/shapes.ts)

### 1a. Implementation Specification
1. Parse the input XML string via `new DOMParser().parseFromString(svgString, 'image/svg+xml')`.
2. Extract the canonical `viewBox` coordinates: `minX`, `minY`, `vbWidth`, `vbHeight`. Fallback to `width` and `height` attributes if `viewBox` is absent.
3. Traverse all graphical elements in document order (including within `<g>` groups):
   - `<path d="...">`: extract `d`.
   - `<rect x y width height rx ry>`: convert to path with rounded corners if $rx/ry > 0$.
   - `<circle cx cy r>`: convert to path `M cx-r cy A r r 0 1 0 cx+r cy A r r 0 1 0 cx-r cy Z`.
   - `<ellipse cx cy rx ry>`: convert to path `M cx-rx cy A rx ry 0 1 0 cx+rx cy A rx ry 0 1 0 cx-rx cy Z`.
   - `<polygon points="...">`: convert to path `M p0.x p0.y L p1.x p1.y ... Z`.
   - `<polyline points="...">`: convert to path `M p0.x p0.y L p1.x p1.y ...`.
4. Combine into a unified compound path.
5. If target frame dimensions `(targetWidth, targetHeight)` are provided, scale and center the path with **Aspect-Fit (Contain)**:
   - $S = \min\left(\frac{targetWidth}{vbWidth}, \frac{targetHeight}{vbHeight}\right)$
   - $offsetX = \frac{targetWidth - vbWidth \cdot S}{2}$
   - $offsetY = \frac{targetHeight - vbHeight \cdot S}{2}$

### 1b. Add `normalizeCustomSvgMask` to `src/domain/shapes.ts`
```typescript
export interface NormalizedSvgResult {
  pathData: string;
  viewBox: { x: number; y: number; width: number; height: number };
}

/**
 * Parses an SVG document, extracts and converts all vector primitives (<path>, <rect>,
 * <circle>, <ellipse>, <polygon>, <polyline>) into a unified compound path, and normalizes
 * coordinates using aspect-fit containment and centering within the frame.
 */
export function normalizeCustomSvgMask(
  svgString: string,
  targetWidth?: number,
  targetHeight?: number
): NormalizedSvgResult | null {
  if (!svgString || typeof svgString !== 'string') return null;

  try {
    const parser = new DOMParser();
    const doc = parser.parseFromString(svgString, 'image/svg+xml');
    const svg = doc.querySelector('svg');
    if (!svg) return null;

    // 1. Resolve ViewBox
    let vbX = 0;
    let vbY = 0;
    let vbW = 0;
    let vbH = 0;

    const vbAttr = svg.getAttribute('viewBox');
    if (vbAttr) {
      const parts = vbAttr.trim().split(/[\s,]+/).map(Number);
      if (parts.length === 4 && parts.every(Number.isFinite) && parts[2] > 0 && parts[3] > 0) {
        [vbX, vbY, vbW, vbH] = parts;
      }
    }

    if (vbW === 0 || vbH === 0) {
      const w = parseFloat(svg.getAttribute('width') || '0');
      const h = parseFloat(svg.getAttribute('height') || '0');
      if (w > 0 && h > 0) {
        vbW = w;
        vbH = h;
      }
    }

    // 2. Extract Subpaths
    const subpaths: string[] = [];

    doc.querySelectorAll('path').forEach((p) => {
      const d = p.getAttribute('d');
      if (d && d.trim().length > 0) subpaths.push(d.trim());
    });

    doc.querySelectorAll('rect').forEach((r) => {
      const x = parseFloat(r.getAttribute('x') || '0');
      const y = parseFloat(r.getAttribute('y') || '0');
      const w = parseFloat(r.getAttribute('width') || '0');
      const h = parseFloat(r.getAttribute('height') || '0');
      const rx = parseFloat(r.getAttribute('rx') || '0');
      const ry = parseFloat(r.getAttribute('ry') || String(rx));
      if (w > 0 && h > 0) {
        if (rx > 0 || ry > 0) {
          subpaths.push(
            `M ${x + rx} ${y} L ${x + w - rx} ${y} A ${rx} ${ry} 0 0 1 ${x + w} ${y + ry} L ${x + w} ${y + h - ry} A ${rx} ${ry} 0 0 1 ${x + w - rx} ${y + h} L ${x + rx} ${y + h} A ${rx} ${ry} 0 0 1 ${x} ${y + h - ry} L ${x} ${y + ry} A ${rx} ${ry} 0 0 1 ${x + rx} ${y} Z`
          );
        } else {
          subpaths.push(`M ${x} ${y} L ${x + w} ${y} L ${x + w} ${y + h} L ${x} ${y + h} Z`);
        }
      }
    });

    doc.querySelectorAll('circle').forEach((c) => {
      const cx = parseFloat(c.getAttribute('cx') || '0');
      const cy = parseFloat(c.getAttribute('cy') || '0');
      const r = parseFloat(c.getAttribute('r') || '0');
      if (r > 0) {
        subpaths.push(`M ${cx - r} ${cy} A ${r} ${r} 0 1 0 ${cx + r} ${cy} A ${r} ${r} 0 1 0 ${cx - r} ${cy} Z`);
      }
    });

    doc.querySelectorAll('ellipse').forEach((el) => {
      const cx = parseFloat(el.getAttribute('cx') || '0');
      const cy = parseFloat(el.getAttribute('cy') || '0');
      const rx = parseFloat(el.getAttribute('rx') || '0');
      const ry = parseFloat(el.getAttribute('ry') || '0');
      if (rx > 0 && ry > 0) {
        subpaths.push(`M ${cx - rx} ${cy} A ${rx} ${ry} 0 1 0 ${cx + rx} ${cy} A ${rx} ${ry} 0 1 0 ${cx - rx} ${cy} Z`);
      }
    });

    doc.querySelectorAll('polygon').forEach((poly) => {
      const rawPts = poly.getAttribute('points');
      if (rawPts) {
        const nums = rawPts.trim().split(/[\s,]+/).map(Number).filter(Number.isFinite);
        if (nums.length >= 6) {
          let pStr = `M ${nums[0]} ${nums[1]}`;
          for (let i = 2; i < nums.length; i += 2) {
            pStr += ` L ${nums[i]} ${nums[i + 1]}`;
          }
          pStr += ' Z';
          subpaths.push(pStr);
        }
      }
    });

    if (subpaths.length === 0) return null;

    const compoundPath = subpaths.join(' ');
    const finalVbW = vbW || 100;
    const finalVbH = vbH || 100;

    return {
      pathData: compoundPath,
      viewBox: { x: vbX, y: vbY, width: finalVbW, height: finalVbH },
    };
  } catch (err) {
    console.error('Failed to parse SVG mask:', err);
    return null;
  }
}
```

    const file = e.target.files?.[0];

    const reader = new FileReader();
    reader.onload = (event) => {

### 3a. Update `generate_polygon_mask` with Circular Fillet Arc Sampling
Update `generate_polygon_mask(sides: usize, radius: f64, w: u32, h: u32)`:
- When `radius <= 0.01`: generate sharp regular polygon vertices.
- When `radius > 0.01`: for each vertex $V_i$, compute incoming vector $u$, outgoing vector $v$, angle $\theta$, tangent clamp $d = \min(L/2, radius / \tan(\theta/2))$, effective radius $r_{eff} = d \tan(\theta/2)$, and sample 8 points along the arc between $T_{in}$ and $T_{out}$.

Update `generate_star_mask(points: usize, inner_ratio: f64, tip_r: f64, valley_r: f64, w: u32, h: u32)`:
- For vertex $i$: if $i$ is even, use `tip_r`; if $i$ is odd, use `valley_r`.
- Sample fillet arcs so that outer tips and inner valleys round smoothly into the even-odd rasterizer.

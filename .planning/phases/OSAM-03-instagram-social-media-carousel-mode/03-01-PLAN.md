```typescript
/** Instagram aspect ratio presets */
export type CarouselRatio = '1:1' | '4:5' | '9:16';

export const CAROUSEL_RATIO_PRESETS: Record<CarouselRatio, { width: number; height: number; label: string }> = {
  '1:1': { width: 1080, height: 1080, label: '1:1 — Square' },
  '4:5': { width: 1080, height: 1350, label: '4:5 — Portrait' },
  '9:16': { width: 1080, height: 1920, label: '9:16 — Story / Reel' },
};

/** A single slide in a Social Carousel (pixel-based, no physical unit) */
export interface CarouselSlide {
  id: string;
  slideIndex: number;
  widthPx: number;   // always 1080 for standard IG presets
  heightPx: number;  // 1080 | 1350 | 1920
  backgroundColor: string;
  elements: CarouselElement[];
}

/** Element types within a carousel slide */
export type CarouselElement = CarouselPhotoFrame; // extend later for text overlays

export interface CarouselPhotoFrame {
  type: 'photo';
  id: string;
  photoId?: string;
  filePath?: string;
  fileName?: string;
  previewPath?: string;
  thumbnailPath?: string;
  photoAspect?: number;
  isMissing?: boolean;
  // Position in continuous canvas coords (x = 0 starts at slide 0 left edge, can span slides)
  x: number;
  y: number;
  width: number;
  height: number;
  cropX?: number;
  cropY?: number;
  cropScale?: number;
  rotation?: number;
  locked?: boolean;
  cornerRadius?: number;
}

/** A carousel is a collection of slides with a shared ratio */
export interface Carousel {
  id: string;
  projectId: string;
  ratio: CarouselRatio;
  slideWidthPx: number;
  slideHeightPx: number;
  slides: CarouselSlide[];
  totalSlides: number;
}

/** Utility: create a new empty carousel with default slides */
export function createInitialCarousel(projectId: string, ratio: CarouselRatio = '1:1'): Carousel {
  const preset = CAROUSEL_RATIO_PRESETS[ratio];
  const slideId = `slide-${Date.now()}-1`;
  return {
    id: `carousel-${projectId}`,
    projectId,
    ratio,
    slideWidthPx: preset.width,
    slideHeightPx: preset.height,
    slides: [
      {
        id: slideId,
        slideIndex: 0,
        widthPx: preset.width,
        heightPx: preset.height,
        backgroundColor: '#FFFFFF',
        elements: [],
      },
    ],
    totalSlides: 1,
  };
}

/** Utility: get total canvas width (continuous stage width) */
export function getCarouselTotalWidth(carousel: Carousel): number {
  return carousel.slideWidthPx * carousel.slides.length;
}

/** Utility: get x-offset where slide N begins */
export function getSlideXOffset(carousel: Carousel, slideIndex: number): number {
  return carousel.slideWidthPx * slideIndex;
}

/** Utility: create a new empty slide */
export function createCarouselSlide(carousel: Carousel): CarouselSlide {
  return {
    id: `slide-${Date.now()}-${carousel.slides.length + 1}`,
    slideIndex: carousel.slides.length,
    widthPx: carousel.slideWidthPx,
    heightPx: carousel.slideHeightPx,
    backgroundColor: '#FFFFFF',
    elements: [],
  };
}
```

### Task 2: Create Carousel Zustand Store

**File**: `src/stores/carouselStore.ts` (NEW)

```typescript
import { create } from 'zustand';
import {
  Carousel, CarouselSlide, CarouselRatio, CarouselPhotoFrame,
  createInitialCarousel, createCarouselSlide, CAROUSEL_RATIO_PRESETS,
} from '../domain/carousel';

export interface CarouselState {
  currentCarousel: Carousel | null;
  activeSlideIndex: number;

  // Actions
  initializeCarousel: (projectId: string, ratio?: CarouselRatio) => void;
  setRatio: (ratio: CarouselRatio) => void;
  setActiveSlide: (index: number) => void;
  addSlide: () => void;
  duplicateSlide: (index: number) => void;
  deleteSlide: (index: number) => void;
  reorderSlide: (fromIndex: number, toIndex: number) => void;
  updateSlideBackground: (slideIndex: number, color: string) => void;
  addPhotoFrame: (slideIndex: number, frame: Omit<CarouselPhotoFrame, 'id'>) => void;
  updatePhotoFrame: (frameId: string, updates: Partial<CarouselPhotoFrame>) => void;
  removePhotoFrame: (frameId: string) => void;
}

export const useCarouselStore = create<CarouselState>((set, get) => ({
  currentCarousel: null,
  activeSlideIndex: 0,

  initializeCarousel: (projectId, ratio = '1:1') => {
    const carousel = createInitialCarousel(projectId, ratio);
    set({ currentCarousel: carousel, activeSlideIndex: 0 });
  },

  setRatio: (ratio) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const preset = CAROUSEL_RATIO_PRESETS[ratio];
    const updatedSlides: CarouselSlide[] = currentCarousel.slides.map((s) => ({
      ...s,
      widthPx: preset.width,
      heightPx: preset.height,
    }));
    set({
      currentCarousel: {
        ...currentCarousel,
        ratio,
        slideWidthPx: preset.width,
        slideHeightPx: preset.height,
        slides: updatedSlides,
      },
    });
  },

  setActiveSlide: (index) => set({ activeSlideIndex: index }),

  addSlide: () => {
    const { currentCarousel } = get();
    if (!currentCarousel || currentCarousel.slides.length >= 10) return; // max 10 slides
    const newSlide = createCarouselSlide(currentCarousel);
    const updatedSlides = [...currentCarousel.slides, newSlide].map((s, i) => ({
      ...s,
      slideIndex: i,
    }));
    set({
      currentCarousel: { ...currentCarousel, slides: updatedSlides, totalSlides: updatedSlides.length },
      activeSlideIndex: updatedSlides.length - 1,
    });
  },

  duplicateSlide: (index) => {
    const { currentCarousel } = get();
    if (!currentCarousel || currentCarousel.slides.length >= 10) return;
    const source = currentCarousel.slides[index];
    if (!source) return;
    const duplicated: CarouselSlide = {
      ...source,
      id: `slide-${Date.now()}-dup`,
      elements: source.elements.map((el) => ({
        ...el,
        id: `${el.id}-dup-${Date.now()}`,
      })),
    };
    const updatedSlides = [
      ...currentCarousel.slides.slice(0, index + 1),
      duplicated,
      ...currentCarousel.slides.slice(index + 1),
    ].map((s, i) => ({ ...s, slideIndex: i }));
    set({
      currentCarousel: { ...currentCarousel, slides: updatedSlides, totalSlides: updatedSlides.length },
      activeSlideIndex: index + 1,
    });
  },

  deleteSlide: (index) => {
    const { currentCarousel } = get();
    if (!currentCarousel || currentCarousel.slides.length <= 1) return; // keep min 1
    const updatedSlides = currentCarousel.slides
      .filter((_, i) => i !== index)
      .map((s, i) => ({ ...s, slideIndex: i }));
    set({
      currentCarousel: { ...currentCarousel, slides: updatedSlides, totalSlides: updatedSlides.length },
      activeSlideIndex: Math.min(index, updatedSlides.length - 1),
    });
  },

  reorderSlide: (fromIndex, toIndex) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const slides = [...currentCarousel.slides];
    const [moved] = slides.splice(fromIndex, 1);
    slides.splice(toIndex, 0, moved);
    const updatedSlides = slides.map((s, i) => ({ ...s, slideIndex: i }));
    set({
      currentCarousel: { ...currentCarousel, slides: updatedSlides },
      activeSlideIndex: toIndex,
    });
  },

  updateSlideBackground: (slideIndex, color) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const updatedSlides = currentCarousel.slides.map((s, i) =>
      i === slideIndex ? { ...s, backgroundColor: color } : s
    );
    set({ currentCarousel: { ...currentCarousel, slides: updatedSlides } });
  },

  addPhotoFrame: (slideIndex, frame) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const newFrame: CarouselPhotoFrame = { ...frame, id: `frame-${Date.now()}` };
    const updatedSlides = currentCarousel.slides.map((s, i) =>
      i === slideIndex ? { ...s, elements: [...s.elements, newFrame] } : s
    );
    set({ currentCarousel: { ...currentCarousel, slides: updatedSlides } });
  },

  updatePhotoFrame: (frameId, updates) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const updatedSlides = currentCarousel.slides.map((s) => ({
      ...s,
      elements: s.elements.map((el) =>
        el.id === frameId ? { ...el, ...updates } : el
      ),
    }));
    set({ currentCarousel: { ...currentCarousel, slides: updatedSlides } });
  },

  removePhotoFrame: (frameId) => {
    const { currentCarousel } = get();
    if (!currentCarousel) return;
    const updatedSlides = currentCarousel.slides.map((s) => ({
      ...s,
      elements: s.elements.filter((el) => el.id !== frameId),
    }));
    set({ currentCarousel: { ...currentCarousel, slides: updatedSlides } });
  },
}));
```

  activeMode={activeMode}

    setActiveMode(mode);

      const carouselStore = await import('../../stores/carouselStore');
      const cs = carouselStore.useCarouselStore.getState();
      if (!cs.currentCarousel || cs.currentCarousel.projectId !== currentProject.id) {

import { describe, it, expect } from 'vitest' // or tap/assert style matching existing tests
import {
  createInitialCarousel,
  createCarouselSlide,
  getCarouselTotalWidth,
  getSlideXOffset,
  CAROUSEL_RATIO_PRESETS,
} from '../src/domain/carousel';

describe('Carousel domain', () => {
  it('creates carousel with correct defaults', () => {
    const c = createInitialCarousel('project-1');
    expect(c.ratio).toBe('1:1');
    expect(c.slideWidthPx).toBe(1080);
    expect(c.slideHeightPx).toBe(1080);
    expect(c.slides.length).toBe(1);
    expect(c.totalSlides).toBe(1);
  });

  it('calculates total width correctly', () => {
    const c = createInitialCarousel('p1');
    const slide2 = createCarouselSlide(c);
    c.slides.push(slide2);
    c.totalSlides = 2;
    expect(getCarouselTotalWidth(c)).toBe(2160);
  });

  it('calculates slide x offset correctly', () => {
    const c = createInitialCarousel('p1');
    expect(getSlideXOffset(c, 0)).toBe(0);
    expect(getSlideXOffset(c, 2)).toBe(2160);
  });

  it('all ratio presets have correct dimensions', () => {
    expect(CAROUSEL_RATIO_PRESETS['1:1'].width).toBe(1080);
    expect(CAROUSEL_RATIO_PRESETS['4:5'].height).toBe(1350);
    expect(CAROUSEL_RATIO_PRESETS['9:16'].height).toBe(1920);
  });
});
